[Heading 4] 模型1：Benford定律异常检测

[First Paragraph] 原理：自然产生的财务数据中，首位数字分布遵循特定规律（1出现约30.1%，9出现约4.6%）。人为造假的数据往往偏离此规律。

[Body Text] 实施步骤：

[Source Code] # 核心算法示例
import numpy as np
from scipy import stats

def benford_test(data):
    """
    检测数据是否符合Benford定律
    data: 金额数据列表
    """
    # 提取首位数字
    first_digits = [int(str(abs(x))[0]) for x in data if x != 0]
    
    # Benford理论分布
    benford_probs = [np.log10(1 + 1/d) for d in range(1, 10)]
    
    # 实际分布
    actual_counts = [first_digits.count(d) for d in range(1, 10)]
    actual_probs = [c / sum(actual_counts) for c in actual_counts]
    
    # 卡方检验
    chi2, p_value = stats.chisquare(actual_counts, 
                                     [sum(actual_counts) * p for p in benford_probs])
    
    return {
        'chi2': chi2,
        'p_value': p_value,
        'is_anomaly': p_value < 0.05,  # 偏离显著则异常
        'actual_dist': actual_probs,
        'benford_dist': benford_probs
    }

def fraud_score(transaction, rules):
    """
    综合风险评分
    transaction: 单笔交易数据
    rules: 触发的规则列表
    """
    score = 0
    weights = {
        'F001': 10, 'F002': 30, 'F003': 15,
        'F004': 25, 'F005': 20, 'F006': 40,
        'F007': 10, 'F008': 35
    }
    
    for rule in rules:
        score += weights.get(rule, 5)
    
    # 风险等级
    if score >= 60: level = '极高'
    elif score >= 40: level = '高'
    elif score >= 20: level = '中'
    else: level = '低'
    
    return {'score': score, 'level': level}

[Source Code] # 预算执行偏差分析
def budget_analysis(budget_data, actual_data):
    """
    分析预算执行异常情况
    """
    results = []
    for item in budget_data:
        budget = item['budget_amount']
        actual = actual_data.get(item['item_code'], 0)
        
        # 执行率
        execution_rate = actual / budget if budget > 0 else 0
        
        # 偏差分析
        deviation = {
            'item_code': item['item_code'],
            'item_name': item['item_name'],
            'budget': budget,
            'actual': actual,
            'execution_rate': execution_rate,
            'deviation_type': None,
            'risk_level': '正常'
        }
        
        # 异常判定
        if execution_rate < 0.5:
            deviation['deviation_type'] = '执行严重不足'
            deviation['risk_level'] = '高'
        elif execution_rate > 1.2:
            deviation['deviation_type'] = '超预算执行'
            deviation['risk_level'] = '高'
        elif execution_rate > 1.0:
            deviation['deviation_type'] = '预算不足'
            deviation['risk_level'] = '中'
        
        # 年底突击花钱检测
        if actual > 0:
            q4_ratio = item.get('q4_actual', 0) / actual
            if q4_ratio > 0.5:
                deviation['year_end_spike'] = True
                deviation['risk_level'] = '高'
        
        results.append(deviation)
    
    return results
